#!/usr/bin/env python3
"""Source-distributed read-only candidate HF/S03 fingerprint. Not owner native attestation."""
import hashlib
import json
import os
from pathlib import Path
import re
import sqlite3
import sys
from urllib.parse import quote

try:
    from safetensors import safe_open
except ImportError:
    raise SystemExit('HOLD_SAFETENSORS_NOT_INSTALLED')

SHA40 = re.compile(r'^[a-f0-9]{40}$')
MAX_FILES = 1000
MAX_INDEX_BYTES = 16 * 1024 * 1024

def hfile(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        while True:
            b=f.read(1024*1024)
            if not b: break
            h.update(b)
    return h.hexdigest()

def within_child(base, name):
    if not isinstance(name,str) or name != Path(name).name or name.startswith('.'):
        raise ValueError('HOLD_INVALID_SHARD_NAME')
    p=base/name
    if not p.is_file() or p.is_symlink():
        raise ValueError('HOLD_MISSING_OR_SYMLINK')
    return p

def fingerprint(student, db):
    if not student.is_dir() or student.is_symlink():
        raise ValueError('HOLD_STUDENT_DIR')
    config_path=within_child(student,'config.json')
    tokconfig_path=within_child(student,'tokenizer_config.json')
    cfg=json.loads(config_path.read_text('utf-8'))
    tok=json.loads(tokconfig_path.read_text('utf-8'))
    shard_list=sorted(p.name for p in student.glob('*.safetensors'))
    if not shard_list or len(shard_list)>MAX_FILES:
        raise ValueError('HOLD_SHARD_COUNT')
    shard_list=[within_child(student,n) for n in shard_list]
    expected_weight_map=None
    index=student/'model.safetensors.index.json'
    if index.exists() or index.is_symlink():
        index=within_child(student,index.name)
        if index.stat().st_size > MAX_INDEX_BYTES:
            raise ValueError('HOLD_SHARD_INDEX_OVERSIZE')
        def distinct_pairs(pairs):
            result={}
            for key,value in pairs:
                if key in result:
                    raise ValueError('HOLD_DUPLICATE_INDEX_KEY')
                result[key]=value
            return result
        info=json.loads(index.read_text('utf-8'),object_pairs_hook=distinct_pairs)
        if not isinstance(info,dict):
            raise ValueError('HOLD_SHARD_INDEX_SCHEMA')
        expected_weight_map=info.get('weight_map')
        if (not isinstance(expected_weight_map,dict) or not expected_weight_map or
            any(not isinstance(k,str) or not k or not isinstance(v,str) or not v
                for k,v in expected_weight_map.items())):
            raise ValueError('HOLD_SHARD_INDEX_SCHEMA')
        refs=set(expected_weight_map.values())
        if refs != set(p.name for p in shard_list):
            raise ValueError('HOLD_SHARD_INDEX_MISMATCH')
    elif len(shard_list)!=1:
        raise ValueError('HOLD_SHARD_INDEX_MISSING')
    filenames=['tokenizer.json','tokenizer.model','vocab.json','merges.txt','special_tokens_map.json','added_tokens.json']
    present={n: hfile(within_child(student,n)) for n in filenames if (student/n).exists()}
    if not set(present)&{'tokenizer.json','tokenizer.model','vocab.json'}:
        raise ValueError('HOLD_TOKENIZER_PAYLOAD_MISSING')
    hashes={p.name:hfile(p) for p in shard_list}
    tensor_records=[]
    observed_weight_map={}
    for p in shard_list:
        with safe_open(str(p),framework='np',device='cpu') as f:
            for name in f.keys():
                if name in observed_weight_map:
                    raise ValueError('HOLD_TENSOR_HEADER_DUP_OR_EMPTY')
                observed_weight_map[name]=p.name
                s=f.get_slice(name)
                tensor_records.append((name,list(s.get_shape()),s.get_dtype()))
    tensor_records.sort(key=lambda r:r[0])
    if not tensor_records:
        raise ValueError('HOLD_TENSOR_HEADER_DUP_OR_EMPTY')
    if expected_weight_map is not None and observed_weight_map!=expected_weight_map:
        raise ValueError('HOLD_SHARD_INDEX_TENSOR_MAP_MISMATCH')
    revision=student.name if SHA40.fullmatch(student.name) and student.parent.name=='snapshots' else None
    if not db.is_file() or db.is_symlink():
        raise ValueError('HOLD_S03_DB')
    uri='file:'+quote(str(db.resolve()))+'?mode=ro'
    con=sqlite3.connect(uri,uri=True,timeout=1)
    try:
        con.execute('PRAGMA query_only=ON')
        schema=con.execute("SELECT type,name,sql FROM sqlite_master WHERE sql IS NOT NULL ORDER BY type,name").fetchall()
        has_s03=con.execute("SELECT 1 FROM sqlite_master WHERE type='table' AND name='s03_items'").fetchone() is not None
        mode=con.execute('PRAGMA journal_mode').fetchone()[0]
        if not has_s03:raise ValueError('HOLD_S03_TABLE_NOT_FOUND')
    finally:
        con.close()
    stable=lambda v: hashlib.sha256(json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False).encode()).hexdigest()
    out={
        'schema':'NOX_2A_LOCAL_READ_ONLY_HF_S03_PROBE_V1',
        'attestation':'UNATTESTED_LOCAL_SNAPSHOT_NOT_NATIVE_OWNER_PROOF',
        'student_revision_from_hf_snapshot_name':revision,
        'revision_verified_from_remote':False,
        'model_type':cfg.get('model_type'),
        'architectures':cfg.get('architectures'),
        'config_sha256':hfile(config_path),
        'tokenizer_config_sha256':hfile(tokconfig_path),
        'chat_template_sha256':stable(tok.get('chat_template')) if 'chat_template' in tok else None,
        'tokenizer_payload_sha256':stable(present),
        'shard_hashes_sha256':stable(hashes),
        'tensor_headers_sha256':stable(tensor_records),
        'tensor_count':len(tensor_records),
        'shard_index_tensor_map_checked':expected_weight_map is not None,
        'shard_index_tensor_map_sha256':stable(observed_weight_map),
        's03_schema_sha256':stable(schema),
        's03_journal_mode':mode,
        's03_rows_read':0,
        's03_writer_lease_verified':False,
        'active_gguf_hf_conversion_verified':False,
        'checkpoint_offsite_restore_verified':False,
        'live_training_identity_verified':False,
        'native_approval_required':True,
    }
    return out

if __name__=='__main__':
    if len(sys.argv)!=3:raise SystemExit('USAGE: python lineage_ro.py STUDENT_DIR S03_DB')
    try:
        print(json.dumps(fingerprint(Path(sys.argv[1]),Path(sys.argv[2])),ensure_ascii=False,sort_keys=True))
    except (OSError,ValueError,sqlite3.Error,json.JSONDecodeError) as e:
        raise SystemExit(str(e) if isinstance(e, ValueError) and str(e).startswith('HOLD_') else 'HOLD_LOCAL_READ_ONLY_PROBE_'+type(e).__name__)
