import re

def audit_pilot(rows):
    allowed = {"Apache-2.0", "CC-BY-4.0"}
    splits = {"train", "holdout", "validation", "test"}
    assigned = {}
    for i, r in enumerate(rows):
        fields = ("repo", "revision", "file_sha256", "licence", "language",
                  "split", "leakage_keys", "privacy_review", "row_review",
                  "payload_sha256")
        if any(k not in r for k in fields):
            raise ValueError(f"row_{i}: missing metadata")
        if not re.fullmatch(r"[0-9a-f]{40}", r["revision"]) or any(
            not re.fullmatch(r"[0-9a-f]{64}", r[k])
            for k in ("file_sha256", "payload_sha256")
        ):
            raise ValueError(f"row_{i}: unpinned provenance")
        if (r["licence"] not in allowed or r["language"] not in {"ru", "es"}
                or r["split"] not in splits):
            raise ValueError(f"row_{i}: unsupported licence/language/split")
        if (r["privacy_review"] != "PASS" or r["row_review"] != "PASS"
                or not r["leakage_keys"]):
            raise ValueError(f"row_{i}: review or grouping HOLD")
        for key in r["leakage_keys"]:
            if not isinstance(key, str) or not key:
                raise ValueError(f"row_{i}: invalid leakage key")
            old = assigned.setdefault(key, r["split"])
            if old != r["split"]:
                raise ValueError(f"row_{i}: cross-split contamination")
    return len(rows)
